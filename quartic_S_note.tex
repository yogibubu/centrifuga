\section*{Goal}
Given
\[
\mathbf D_S=(D_J,D_{JK},D_K,d_1,d_2)^\mathsf T,
\]
derive the representation transforms among $I,II,III$ at fixed S reduction:
\[
\mathbf D_{S,R'} = T_S(R\to R')\,\mathbf D_{S,R}.
\]

\section*{Procedure (implemented)}
Start from the quartic Watson Hamiltonian in S reduction,
\[
H_4=-D_JJ^4-D_{JK}J^2J_z^2-D_KJ_z^4-d_1J^2(J_x^2-J_y^2)-d_2(J_x^4-J_y^4),
\]
expand it in each representation $R\in\{I,II,III\}$ using the cyclic axis maps, and collect quartic monomials in $(J_a,J_b,J_c)$.

Define a determinable vector
\[
\mathbf T=(T_{aa},T_{bb},T_{cc},T_1,T_2)^\mathsf T,
\]
with
\[
T_{aa}=\mathrm{coeff}(J_a^4),\quad
T_{bb}=\mathrm{coeff}(J_b^4),\quad
T_{cc}=\mathrm{coeff}(J_c^4),
\]
\[
T_1=\frac12\left[\mathrm{coeff}(J_a^2J_b^2)+\mathrm{coeff}(J_a^2J_c^2)+\mathrm{coeff}(J_b^2J_c^2)\right],
\]
\[
T_2=A\,\mathrm{coeff}(J_a^2J_b^2)+B\,\mathrm{coeff}(J_a^2J_c^2)+C\,\mathrm{coeff}(J_b^2J_c^2).
\]
Then
\[
\mathbf T = Q_S(R)\,\mathbf D_{S,R},
\qquad
T_S(R\to R') = Q_S(R')^{-1}Q_S(R).
\]
