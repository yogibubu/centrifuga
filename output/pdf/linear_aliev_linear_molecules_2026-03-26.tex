\section*{2. Indices and objects}

\begin{itemize}
\item Parallel modes are indexed by $n,n',n'',\dots$.
\item Perpendicular degenerate pairs are indexed by $t,t',\dots$.
\item $B$ is the equilibrium rotational constant.
\item $D_J$ is the equilibrium quartic centrifugal distortion constant.
\item $\omega_n$ and $\omega_t$ are harmonic frequencies of parallel and
perpendicular modes.
\item $B_n^{xx}$ is the parallel rotational derivative entering
$C_n = -B B_n^{xx}/\omega_n$.
\item $k'_{ijk}$ and $k'_{ijkl}$ denote the cubic and quartic constants in the
Aliev-side coordinate convention used by the implementation.
\end{itemize}

\section*{3. Primary definitions}

The implemented primary objects are
\begin{align}
C_n &= - \frac{B B_n^{xx}}{\omega_n},
\\[0.4em]
X^{nt} &= \frac{2B\zeta_{nt}\sqrt{\omega_n\omega_t}}{\omega_n^2-\omega_t^2},
\\[0.4em]
X_{nt} &= \frac{B\zeta_{nt}(\omega_n^2+\omega_t^2)}
{\sqrt{\omega_n\omega_t}(\omega_n^2-\omega_t^2)},
\\[0.4em]
r_n &= 4\omega_n C_n\left(\frac{D_J}{B}-\frac{B^2}{\omega_n^2}\right)
+ \frac{1}{2}\sum_{n',n''} k'_{nn'n''} C_{n'} C_{n''},
\\[0.4em]
r_{nn'} &= \frac{3\omega_n\omega_{n'} C_n C_{n'}}{4B}
+ \frac{1}{2}\sum_{n''} k'_{nn'n''} C_{n''},
\\[0.4em]
r_{tt'} &= \frac{1}{2}\sum_n k'_{tt'n} C_n.
\end{align}

\section*{4. Auxiliary families}

The implemented $F$, $U$, and $V$ families are
\begin{align}
F^{nn'} &=
B^2\sum_t \zeta_{nt}\zeta_{n't}\sqrt{\omega_n\omega_{n'}}
\frac{\omega_n^2+\omega_{n'}^2-2\omega_t^2}
{(\omega_n^2-\omega_t^2)(\omega_{n'}^2-\omega_t^2)},
\\[0.6em]
F_{nn'} &=
r_{nn'}
+ B^2\sum_t \zeta_{nt}\zeta_{n't}
\frac{\omega_n^2\omega_{n'}^2-\omega_t^4}
{\omega_t\sqrt{\omega_n\omega_{n'}}(\omega_n^2-\omega_t^2)(\omega_{n'}^2-\omega_t^2)},
\\[0.6em]
U_{nn'} &=
\frac{r_{nn'}}{4(\omega_n+\omega_{n'})}
+ \frac{B^2}{8}\sum_t
\frac{\zeta_{nt}\zeta_{n't}}
{\omega_t(\omega_n\omega_{n'})^{1/2}}
\frac{(\omega_n+\omega_t)^2(\omega_{n'}-\omega_t)^2+(\omega_n-\omega_t)^2(\omega_{n'}+\omega_t)^2}
{(\omega_n^2-\omega_t^2)(\omega_{n'}^2-\omega_t^2)},
\\[0.6em]
V_{nn'} &=
\frac{r_{nn'}}{4(\omega_n-\omega_{n'})}
+ \frac{B^2}{8}\sum_t
\frac{\zeta_{nt}\zeta_{n't}}
{\omega_t(\omega_n\omega_{n'})^{1/2}}
\frac{
(\omega_n+\omega_t)^2(\omega_{n'}+\omega_t)^2(\omega_n+\omega_{n'}-2\omega_t)
-
(\omega_n-\omega_t)^2(\omega_{n'}-\omega_t)^2(\omega_n+\omega_{n'}+2\omega_t)
}
{(\omega_n^2-\omega_t^2)(\omega_{n'}^2-\omega_t^2)(\omega_n-\omega_{n'})}.
\end{align}

For the perpendicular branch the implementation also keeps the analogous
families
\begin{align}
U_{tt'} &=
\frac{r_{tt'}}{4(\omega_t+\omega_{t'})}
+ \frac{B^2}{8}\sum_n
\frac{\zeta_{nt}\zeta_{nt'}}
{\omega_n(\omega_t\omega_{t'})^{1/2}}
\frac{(\omega_t+\omega_n)^2(\omega_{t'}-\omega_n)^2+(\omega_t-\omega_n)^2(\omega_{t'}+\omega_n)^2}
{(\omega_t^2-\omega_n^2)(\omega_{t'}^2-\omega_n^2)},
\\[0.6em]
V_{tt'} &=
\frac{r_{tt'}}{4(\omega_t-\omega_{t'})}
+ \frac{B^2}{8}\sum_n
\frac{\zeta_{nt}\zeta_{nt'}}
{\omega_n(\omega_t\omega_{t'})^{1/2}}
\frac{
(\omega_t+\omega_{t'})^2(\omega_{t'}+\omega_n)^2(\omega_t+\omega_{t'}-2\omega_n)
-
(\omega_t-\omega_{t'})^2(\omega_{t'}-\omega_n)^2(\omega_t+\omega_{t'}+2\omega_n)
}
{(\omega_t^2-\omega_n^2)(\omega_{t'}^2-\omega_n^2)(\omega_t-\omega_{t'})}.
\end{align}

\section*{5. Parallel coefficient \texorpdfstring{$\beta_n$}{beta n}}

The implemented parallel coefficient is
\begin{equation}
\begin{aligned}
\beta_n &=
4\omega_n^2 C_n^2\left(\frac{B}{\omega_n^2}-\frac{D_J}{B^2}\right)
+ \frac{1}{4}\sum_{n',n''} k^{(4)}_{\beta_n}(n;n',n'') C_{n'}C_{n''}
- \sum_{n'} \frac{k'_{nnnn'}}{2\omega_{n'}} r_{n'}
\\[0.4em]
&\quad
- 4B\sum_{n',t} C_{n'}\zeta_{n't}
\frac{
(3\omega_n^2+\omega_t^2)\omega_n^{3/2} C_n \zeta_{nt}
+
(\omega_n^2+\omega_t^2)\omega_n^{3/2} C_n r_{nn'}
}
{\sqrt{\omega_n\omega_t}(\omega_n^2-\omega_t^2)}
\\[0.4em]
&\quad
+ 4\sum_{n',t}\left(X_{nt}X_{n't}F^{nn'} + X^{nt}X^{n't}F_{nn'}\right)
- 4\sum_{n',t}\left(X_{nt}X^{n't}F_{n'n} + X^{nt}X_{n't}F^{n'n}\right)
\\[0.4em]
&\quad
- 16\sum_{n'}
\left[U_{nn'}(\omega_n+\omega_{n'}) + V_{nn'}(\omega_{n'}-\omega_n)\right].
\end{aligned}
\end{equation}

\paragraph{Final working convention for the current linearmolecule branch.}
In the current working branch, the final parallel $uv$ insertion is disabled:
\[
(\beta_n)_{\mathrm{frozen}}
=
(\beta_n)_{\mathrm{full}}
- \left[-16\sum_{n'}
\left( U_{nn'}(\omega_n+\omega_{n'}) + V_{nn'}(\omega_{n'}-\omega_n)\right)\right].
\]

This is not presented as a literature-derived theorem. It is the final working
convention of the present linear branch because, with the current Gaussian
$\to$ Aliev mapping, the parallel $uv$ block behaves as a dominant non-modal
offset and drives otherwise sane parallel outputs to the wrong scale.

\section*{6. Perpendicular coefficient \texorpdfstring{$\beta_t$}{beta t}}

The implemented perpendicular coefficient is
\begin{equation}
\begin{aligned}
\beta_t &=
\frac{1}{4}\sum_{n,n'} k^{(4)}_{\beta_t}(t;n,n') C_n C_{n'}
+ \mathcal{P}_t
- \sum_n \frac{k'_{ttn}}{2\omega_n} r_n
\\[0.4em]
&\quad
- 4B\sum_{n,n'} C_n \zeta_{nt}
\frac{
2\omega_t^2 \omega_{n'}^{3/2} C_{n'}\zeta_{n't}
+
(3\omega_t^2+\omega_{n'}^2)\omega_{n'}^{3/2} C_{n'} r_{nn'}
}
{\omega_t\sqrt{\omega_{n'}}(\omega_t^2-\omega_{n'}^2)}
\\[0.4em]
&\quad
+ 4\sum_{n,n'}\left(X_{nt}X_{n't}F^{nn'} + X^{nt}X^{n't}F_{nn'}\right)
- 4\sum_{n,t'}\left(X_{nt}X^{nt'}F_{tt'} + X^{nt}X_{nt'}F^{tt'}\right)
\\[0.4em]
&\quad
- 16\sum_{t'} \left[U_{tt'}(\omega_t+\omega_{t'}) + V_{tt'}(\omega_{t'}-\omega_t)\right].
\end{aligned}
\end{equation}

\section*{7. Optical scalar \texorpdfstring{$L$}{L}}

The implemented quartic optical scalar is
\begin{equation}
L =
-\frac{8D_J^3}{B^2}
+ \frac{1}{24}\sum_{n,n',n'',n'''} k'_{nn'n''n'''} C_n C_{n'} C_{n''} C_{n'''}
+ 8BD_J \sum_n \frac{C_n^2}{\omega_n}
- \sum_n \frac{r_n^2}{2\omega_n}.
\end{equation}

This is exposed programmatically by
\[
\texttt{build\_explicit\_aliev\_L\_model(inputs).value}.
\]

The implementation also provides a bookkeeping partition of $L$ into modewise
contributions, but this partition remains provisional because the global
$-8D_J^3/B^2$ term is not intrinsically mode-local.

\section*{8. Final state dependence}

The state dependence implemented for linear molecules is
\begin{equation}
D_v = D_J - \sum_n \beta_n\left(v_n+\frac{1}{2}\right) - \sum_t \beta_t (v_t+1).
\end{equation}

The current linear branch should be read in the following way.

\begin{itemize}
\item The algebraic implementation of the linear families, of $\beta_n$,
$\beta_t$, and of the optical scalar $L$ is now explicit and internally

if it were already the physical Aliev $\zeta$ object.

benchmark branch.
\item It is still not a fully derived, literature-validated final theory branch,

  \begin{itemize}
  \item the sign of the perpendicular $xf/cross$ insertion;
  \item the exclusion of the parallel $uv$ insertion.
  \end{itemize}
\end{itemize}

So the correct conclusion is:

\begin{quote}
The linear-molecule branch is now closed as a strong working implementation,
including the optical scalar $L$, but not yet closed as a fully justified final
derivation of every remaining convention.
